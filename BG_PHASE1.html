System Architecture

Three personas, two interview types (mock + proctored), one codebase. Next.js handles both frontend and backend via App Router. Every AI call is server-side. No AI keys ever reach the browser.

Browser — Candidate / Recruiter
Next.js App Router Tailwind CSS Clerk (auth UI) Stream Video SDK Deepgram WS (STT) face-api.js (CV) Monaco Editor
API Layer — Next.js Route Handlers (Server)
/api/resume/* /api/interview/* /api/proctor/* /api/evaluation/* /api/recruiter/* /api/jobs/process
AI & Async Services
Claude API (Anthropic) Custom Job Queue (Postgres) Stream Video API Clerk (JWT verify)
Data Layer — Supabase
PostgreSQL (RLS) pgvector (embeddings) Storage (PDF, reports) Realtime (recruiter live)
Key Decision All Claude API, and Supabase service-role calls happen exclusively in Next.js Route Handlers (server-side). The browser never touches these keys. Deepgram is the only exception — its key is exposed via NEXT_PUBLIC_DEEPGRAM_API_KEY for the live WebSocket stream, which is acceptable as Deepgram keys are usage-limited. Background jobs run via a custom Postgres-backed queue — no third-party job service needed.

Database Schema — Supabase / PostgreSQL

Run this SQL in the Supabase SQL editor. Enable pgvector first. All tables use Row Level Security (RLS) — users only see their own data.

sql supabase/schema.sql 
-- Enable pgvector extension for embeddings
create extension if not exists vector;

-- ── Users ───────────────────────────────────────────
create table users (
  id          uuid primary key default gen_random_uuid(),
  clerk_id    text unique not null,
  email       text not null,
  name        text,
  role        text check (role in ('candidate', 'recruiter')) not null default 'candidate',
  created_at  timestamptz default now()
);

-- ── Resumes ─────────────────────────────────────────
create table resumes (
  id             uuid primary key default gen_random_uuid(),
  user_id        uuid references users(id) on delete cascade,
  storage_path   text not null,       -- Supabase Storage path
  filename       text,
  extracted_text text,
  skills         jsonb default '[]',  -- Claude-extracted skills array
  created_at     timestamptz default now()
);

-- ── Resume Chunks (for RAG) ──────────────────────────
create table resume_chunks (
  id          uuid primary key default gen_random_uuid(),
  resume_id   uuid references resumes(id) on delete cascade,
  content     text not null,
  embedding   vector(1536),       -- Claude embedding dimension
  chunk_index int,
  created_at  timestamptz default now()
);

-- ── Job Descriptions ────────────────────────────────
create table job_descriptions (
  id             uuid primary key default gen_random_uuid(),
  user_id        uuid references users(id) on delete cascade,
  title          text,
  company        text,
  content        text not null,
  skills_required jsonb default '[]',
  created_at     timestamptz default now()
);

-- ── Interview Sessions ───────────────────────────────
create table sessions (
  id              uuid primary key default gen_random_uuid(),
  candidate_id    uuid references users(id),
  recruiter_id    uuid references users(id),
  resume_id       uuid references resumes(id),
  jd_id           uuid references job_descriptions(id),
  type            text check (type in ('mock', 'proctored')) not null,
  status          text check (status in ('pending','active','completed','terminated')) default 'pending',
  stream_room_id  text,           -- Stream Video room (proctored only)
  warning_count   int default 0,
  started_at      timestamptz,
  ended_at        timestamptz,
  created_at      timestamptz default now()
);

-- ── Conversation Messages ────────────────────────────
create table messages (
  id               uuid primary key default gen_random_uuid(),
  session_id       uuid references sessions(id) on delete cascade,
  role             text check (role in ('ai', 'candidate')) not null,
  content          text not null,
  transcript_conf  float,         -- Deepgram confidence score
  audio_duration   float,         -- seconds
  created_at       timestamptz default now()
);

-- ── Proctoring Events ────────────────────────────────
create table proctor_events (
  id            uuid primary key default gen_random_uuid(),
  session_id    uuid references sessions(id) on delete cascade,
  event_type    text not null,   -- 'no_face','multiple_faces','gaze_off','tab_switch'
  severity      text check (severity in ('warning', 'severe')),
  warning_num   int,              -- 1, 2, or 3
  metadata      jsonb,
  created_at    timestamptz default now()
);

-- ── Evaluations ──────────────────────────────────────
create table evaluations (
  id                  uuid primary key default gen_random_uuid(),
  session_id          uuid references sessions(id) on delete cascade,
  technical_score     int check (technical_score between 0 and 100),
  communication_score int check (communication_score between 0 and 100),
  confidence_score    int check (confidence_score between 0 and 100),
  behavioral_score    int check (behavioral_score between 0 and 100),
  strengths           jsonb default '[]',
  weaknesses          jsonb default '[]',
  improvements        jsonb default '[]',
  detailed_feedback   text,
  report_path         text,       -- Supabase Storage path to PDF
  created_at          timestamptz default now()
);

-- ── Vector similarity search function ───────────────
create or replace function match_resume_chunks(
  query_embedding vector(1536),
  match_threshold float default 0.75,
  match_count     int   default 5,
  resume_id_filter uuid  default null
) returns table (id uuid, content text, similarity float)
language sql stable as $$
  select id, content,
    1 - (embedding <-> query_embedding) as similarity
  from resume_chunks
  where (resume_id_filter is null or resume_id = resume_id_filter)
    and 1 - (embedding <-> query_embedding) > match_threshold
  order by embedding <-> query_embedding
  limit match_count;
$$;

-- ── Enable Row Level Security ────────────────────────
alter table users           enable row level security;
alter table resumes         enable row level security;
alter table sessions        enable row level security;
alter table messages        enable row level security;
alter table evaluations     enable row level security;
alter table proctor_events  enable row level security;

Backend API Routes

Resume Upload + Parse + Embed

typescript app/api/resume/upload/route.ts 
import { auth } from '@clerk/nextjs'
import { createClient } from '@/lib/supabase/server'
import { extractSkills, embedChunks, chunkText } from '@/lib/embeddings'
import pdf from 'pdf-parse'

export async function POST(req: Request) {
  const { userId } = auth()
  if (!userId) return Response.json({ error: 'Unauthorized' }, { status: 401 })

  const formData = await req.formData()
  const file = formData.get('resume') as File
  const supabase = createClient()

  // 1. Upload PDF to Supabase Storage
  const path = `resumes/${userId}/${Date.now()}.pdf`
  await supabase.storage.from('documents').upload(path, file)

  // 2. Extract text from PDF
  const buffer = Buffer.from(await file.arrayBuffer())
  const { text } = await pdf(buffer)

  // 3. Extract skills via Claude
  const skills = await extractSkills(text)

  // 4. Save resume record
  const { data: resume } = await supabase.from('resumes')
    .insert({ user_id: userId, storage_path: path, extracted_text: text, skills })
    .select().single()

  // 5. Chunk text + generate embeddings + store in pgvector
  const chunks = chunkText(text, 500) // 500 tokens per chunk
  await embedChunks(chunks, resume.id, supabase)

  return Response.json({ resumeId: resume.id, skills })
}

Create Interview Session

typescript app/api/interview/create/route.ts 
import { auth } from '@clerk/nextjs'
import { createClient } from '@/lib/supabase/server'
import { streamClient } from '@/lib/stream'

export async function POST(req: Request) {
  const { userId } = auth()
  const { resumeId, jdId, type } = await req.json()
  const supabase = createClient()

  // Create session row
  const { data: session } = await supabase.from('sessions')
    .insert({ candidate_id: userId, resume_id: resumeId, jd_id: jdId, type, status: 'active', started_at: new Date() })
    .select().single()

  let streamToken = null
  let streamRoomId = null

  if (type === 'proctored') {
    // Create Stream Video room + generate user token
    const call = streamClient.video.call('default', session.id)
    await call.getOrCreate()
    streamToken = streamClient.generateUserToken({ user_id: userId })
    streamRoomId = session.id

    await supabase.from('sessions')
      .update({ stream_room_id: streamRoomId })
      .eq('id', session.id)
  }

  return Response.json({ sessionId: session.id, streamToken, streamRoomId })
}

Proctoring Event + Warning Logic

typescript app/api/proctor/event/route.ts 
export async function POST(req: Request) {
  const { sessionId, eventType, metadata } = await req.json()
  const supabase = createClient()

  // Increment warning count
  const { data: session } = await supabase.from('sessions')
    .update({ warning_count: supabase.rpc('increment', { x: 1 }) })
    .eq('id', sessionId).select('warning_count, stream_room_id').single()

  const warningNum = session.warning_count
  const severity = warningNum >= 2 ? 'severe' : 'warning'

  // Log the event
  await supabase.from('proctor_events').insert({
    session_id: sessionId, event_type: eventType,
    severity, warning_num: warningNum, metadata
  })

  // 3 warnings = auto-terminate
  if (warningNum >= 3) {
    await supabase.from('sessions')
      .update({ status: 'terminated', ended_at: new Date() }).eq('id', sessionId)
    // Terminate Stream room
    if (session.stream_room_id) {
      const call = streamClient.video.call('default', session.stream_room_id)
      await call.end()
    }
  }

  return Response.json({ warningNum, severity, terminated: warningNum >= 3 })
}
Warning Counter Bug Use Supabase RPC with a Postgres function to increment warning_count atomically. Never do read-then-write in application code — race conditions will under-count warnings in concurrent requests.

AI Integration — Claude + RAG

Embedding + chunking helper

typescript lib/embeddings.ts 
import Anthropic from '@anthropic-ai/sdk'

const anthropic = new Anthropic()

export function chunkText(text: string, chunkSize = 500): string[] {
  const sentences = text.split(/(?<=[.!?])\s+/)
  const chunks: string[] = []
  let current = ''
  for (const s of sentences) {
    if ((current + s).split(' ').length > chunkSize) {
      chunks.push(current.trim())
      current = s
    } else { current += ' ' + s }
  }
  if (current) chunks.push(current.trim())
  return chunks
}

export async function embedChunks(chunks: string[], resumeId: string, supabase) {
  for (let i = 0; i < chunks.length; i++) {
    const res = await anthropic.embeddings.create({
      model: 'voyage-3', input: chunks[i]  // Anthropic's Voyage embeddings
    })
    await supabase.from('resume_chunks').insert({
      resume_id: resumeId, content: chunks[i],
      embedding: res.embeddings[0].embedding, chunk_index: i
    })
  }
}

export async function extractSkills(resumeText: string): Promise<string[]> {
  const msg = await anthropic.messages.create({
    model: 'claude-sonnet-4-6', max_tokens: 500,
    messages: [{ role: 'user', content: `Extract a JSON array of technical skills from this resume. Return only the JSON array, nothing else.\n\nResume:\n${resumeText}` }]
  })
  return JSON.parse(msg.content[0].text)
}

RAG Question Generation (streaming)

typescript app/api/interview/question/route.ts 
import Anthropic from '@anthropic-ai/sdk'
import { createClient } from '@/lib/supabase/server'

const anthropic = new Anthropic()

export async function POST(req: Request) {
  const { sessionId, resumeId, jdContent, history, questionNum } = await req.json()
  const supabase = createClient()

  // 1. Get query embedding for current interview context
  const queryText = history.length > 0
    ? history[history.length - 1].content  // embed last answer
    : jdContent.slice(0, 200)              // first question: embed JD

  const embRes = await anthropic.embeddings.create({ model: 'voyage-3', input: queryText })
  const embedding = embRes.embeddings[0].embedding

  // 2. Retrieve top-5 relevant resume chunks
  const { data: chunks } = await supabase
    .rpc('match_resume_chunks', { query_embedding: embedding, resume_id_filter: resumeId, match_count: 5 })

  const resumeContext = chunks.map(c => c.content).join('\n---\n')

  // 3. Build conversation history for Claude
  const messages = history.map(m => ({
    role: m.role === 'ai' ? 'assistant' : 'user',
    content: m.content
  }))

  // 4. Stream the next question from Claude
  const stream = await anthropic.messages.stream({
    model: 'claude-sonnet-4-6', max_tokens: 300,
    system: `You are a professional technical interviewer. Use the candidate's resume and job description to ask targeted, relevant questions.
Resume context: ${resumeContext}
Job description: ${jdContent}
Question number: ${questionNum} of 10.
Rules: Ask one focused question. If this is not question 1, refer to the candidate's previous answer. Ask a follow-up or a new topic based on JD requirements. Never repeat questions.`,
    messages: messages.length > 0 ? messages : [{ role: 'user', content: 'Please start the interview.' }]
  })

  return new Response(stream.toReadableStream(), {
    headers: { 'Content-Type': 'text/plain; charset=utf-8' }
  })
}

Speech-to-Text — Deepgram (Live)

Custom hook — useDeepgram

typescript hooks/useDeepgram.ts 
import { useState, useRef, useCallback } from 'react'
import { createClient, LiveTranscriptionEvents } from '@deepgram/sdk'

export function useDeepgram(onTranscript: (text: string, isFinal: boolean) => void) {
  const [isListening, setIsListening] = useState(false)
  const connectionRef = useRef(null)
  const mediaRecorderRef = useRef(null)

  const startListening = useCallback(async () => {
    const deepgram = createClient(process.env.NEXT_PUBLIC_DEEPGRAM_API_KEY)
    const connection = deepgram.listen.live({
      model: 'nova-2', language: 'en-US',
      punctuate: true, interim_results: true,
      utterance_end_ms: 1000,  // fire final after 1s silence
      endpointing: 500
    })

    connection.on(LiveTranscriptionEvents.Transcript, (data) => {
      const transcript = data.channel.alternatives[0].transcript
      if (transcript) onTranscript(transcript, data.is_final)
    })

    connectionRef.current = connection
    const stream = await navigator.mediaDevices.getUserMedia({ audio: true })
    const recorder = new MediaRecorder(stream, { mimeType: 'audio/webm' })
    recorder.addEventListener('dataavailable', (e) => {
      if (connection.getReadyState() === 1) connection.send(e.data)
    })
    recorder.start(250) // send audio chunks every 250ms
    mediaRecorderRef.current = recorder
    setIsListening(true)
  }, [onTranscript])

  const stopListening = useCallback(() => {
    mediaRecorderRef.current?.stop()
    connectionRef.current?.finish()
    setIsListening(false)
  }, [])

  return { isListening, startListening, stopListening }
}

Background Jobs — Custom Postgres Queue

No third-party job service. A jobs table in Supabase acts as the queue. When an interview ends, a row is inserted. A Vercel cron route runs every minute, picks up pending jobs, and processes them. Simple, observable, zero new vendor.

Why build your own instead of a third-party job service You already have Supabase. A Postgres-backed queue is 100 lines of code, has no extra API keys, costs nothing, and you can inspect the queue directly in the Supabase dashboard. Third-party job services solve the same problem but add a vendor dependency for something that isn't genuinely hard.

Step 1 — Add jobs table to your schema

sql supabase/schema.sql (add this table) 
create table jobs (
  id           uuid primary key default gen_random_uuid(),
  type         text not null,             -- 'evaluate_interview'
  payload      jsonb not null,            -- { sessionId: '...' }
  status       text default 'pending'    -- pending | processing | done | failed
               check (status in ('pending','processing','done','failed')),
  attempts     int default 0,
  max_attempts int default 3,
  error        text,
  created_at   timestamptz default now(),
  run_at       timestamptz default now()  -- allows delayed jobs
);

import { createClient } from '@/lib/supabase/server'

export type JobType = 'evaluate_interview' | 'send_report_email'

export async function enqueue(type: JobType, payload: Record<string, unknown>) {
  const supabase = createClient()
  const { error } = await supabase.from('jobs').insert({ type, payload })
  if (error) throw new Error(`Failed to enqueue job: ${error.message}`)
}

Step 4 — Job processor (the actual evaluation logic)

typescript lib/jobs/evaluate.ts 
import Anthropic from '@anthropic-ai/sdk'
import { createClient } from '@/lib/supabase/server'
import { EVALUATION_PROMPT } from '@/lib/prompts'

export async function runEvaluateInterview(payload: { sessionId: string }) {
  const { sessionId } = payload
  const supabase = createClient()
  const anthropic = new Anthropic()

  // 1. Fetch full transcript + JD
  const { data: msgs } = await supabase.from('messages')
    .select('*').eq('session_id', sessionId).order('created_at')

  const { data: session } = await supabase.from('sessions')
    .select('job_descriptions(content)').eq('id', sessionId).single()

  const transcript = msgs.map(m => `${m.role.toUpperCase()}: ${m.content}`).join('\n')
  const jdContent = session.job_descriptions.content

  // 2. Claude evaluation
  const response = await anthropic.messages.create({
    model: 'claude-sonnet-4-6', max_tokens: 2000,
    messages: [{ role: 'user', content: EVALUATION_PROMPT(transcript, jdContent) }]
  })
  const evaluation = JSON.parse(response.content[0].text)

  // 3. Save scores to DB
  await supabase.from('evaluations').insert({ session_id: sessionId, ...evaluation })

  // 4. Broadcast to recruiter via Supabase Realtime
  await supabase.channel(`session-${sessionId}`)
    .send({ type: 'broadcast', event: 'evaluation-ready', payload: evaluation })
}

Step 5 — Cron route (Vercel runs this every minute)

typescript app/api/jobs/process/route.ts 
import { createClient } from '@/lib/supabase/server'
import { runEvaluateInterview } from '@/lib/jobs/evaluate'
import { NextRequest } from 'next/server'

export async function GET(req: NextRequest) {
  // Protect with a secret so only Vercel cron can call this
  const authHeader = req.headers.get('authorization')
  if (authHeader !== `Bearer ${process.env.CRON_SECRET}`) {
    return Response.json({ error: 'Unauthorized' }, { status: 401 })
  }

  const supabase = createClient()

  // Claim up to 5 pending jobs atomically
  const { data: jobs } = await supabase.from('jobs')
    .select('*')
    .eq('status', 'pending')
    .lte('run_at', new Date().toISOString())
    .lt('attempts', supabase.raw('max_attempts'))
    .order('created_at')
    .limit(5)

  for (const job of jobs ?? []) {
    // Mark as processing
    await supabase.from('jobs')
      .update({ status: 'processing', attempts: job.attempts + 1 })
      .eq('id', job.id)

    try {
      if (job.type === 'evaluate_interview') {
        await runEvaluateInterview(job.payload)
      }
      await supabase.from('jobs').update({ status: 'done' }).eq('id', job.id)
    } catch (err) {
      const failed = job.attempts + 1 >= job.max_attempts
      await supabase.from('jobs').update({
        status: failed ? 'failed' : 'pending',
        error: err.message,
        run_at: new Date(Date.now() + 30000).toISOString()  // retry in 30s
      }).eq('id', job.id)
    }
  }

  return Response.json({ processed: jobs?.length ?? 0 })
}

Interview page — full state machine

tsx app/(candidate)/interview/[id]/page.tsx 
'use client'
import { useState, useEffect, useCallback } from 'react'
import { useDeepgram } from '@/hooks/useDeepgram'
import { useProctor } from '@/hooks/useProctor'

type InterviewState = 'intro' | 'questioning' | 'answering' | 'processing' | 'done'

export default function InterviewPage({ params }) {
  const [state, setState] = useState<InterviewState>('intro')
  const [messages, setMessages] = useState([])
  const [currentQ, setCurrentQ] = useState('')
  const [qNum, setQNum] = useState(0)
  const [liveTranscript, setLiveTranscript] = useState('')
  const [warnings, setWarnings] = useState(0)
  const MAX_QUESTIONS = 10

  // Deepgram STT hook
  const { isListening, startListening, stopListening } = useDeepgram(
    useCallback((text, isFinal) => {
      setLiveTranscript(text)
      if (isFinal) handleAnswerFinal(text)
    }, [])
  )

  // Proctoring hook
  useProctor(useCallback(async (eventType) => {
    const res = await fetch('/api/proctor/event', {
      method: 'POST', headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ sessionId: params.id, eventType })
    })
    const { warningNum, terminated } = await res.json()
    setWarnings(warningNum)
    if (terminated) setState('done')
  }, [params.id]))

  async function fetchNextQuestion() {
    setState('questioning')
    const res = await fetch('/api/interview/question', {
      method: 'POST', headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ sessionId: params.id, history: messages, questionNum: qNum + 1 })
    })
    // Stream the question text word by word
    const reader = res.body.getReader()
    let question = ''
    while (true) {
      const { done, value } = await reader.read()
      if (done) break
      question += new TextDecoder().decode(value)
      setCurrentQ(question)
    }
    setQNum(n => n + 1)
    setMessages(m => [...m, { role: 'ai', content: question }])
    setState('answering')
    startListening() // auto-start recording after question
  }

  async function handleAnswerFinal(text: string) {
    stopListening()
    await fetch('/api/interview/message', {
      method: 'POST', headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ sessionId: params.id, role: 'candidate', content: text })
    })
    setMessages(m => [...m, { role: 'candidate', content: text }])
    if (qNum >= MAX_QUESTIONS) { endInterview(); return; }
    fetchNextQuestion()
  }

  async function endInterview() {
    setState('done')
    await fetch('/api/interview/end', {
      method: 'POST', headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ sessionId: params.id })
    })
  }

  return <InterviewLayout state={state} messages={messages} currentQ={currentQ}
    qNum={qNum} max={MAX_QUESTIONS} warnings={warnings} liveTranscript={liveTranscript}
    onStart={fetchNextQuestion} isListening={isListening} />
}

      <StreamCall call={call}>
        <SpeakerLayout />
        <CallControls />
      </StreamCall>

  )

Proctoring — face-api.js

typescript hooks/useProctor.ts 
import { useEffect, useRef } from 'react'
import * as faceapi from 'face-api.js'

export function useProctor(onViolation: (eventType: string) => void) {
  const videoRef = useRef<HTMLVideoElement>(null)
  const intervalRef = useRef(null)

  useEffect(() => {
    async function init() {
      // Load face-api models from /public/models/
      await Promise.all([
        faceapi.nets.tinyFaceDetector.loadFromUri('/models'),
        faceapi.nets.faceLandmark68Net.loadFromUri('/models'),
        faceapi.nets.faceExpressionNet.loadFromUri('/models')
      ])

      const stream = await navigator.mediaDevices.getUserMedia({ video: true })
      if (videoRef.current) videoRef.current.srcObject = stream

      // Check every 1 second
      intervalRef.current = setInterval(async () => {
        if (!videoRef.current) return
        const detections = await faceapi
          .detectAllFaces(videoRef.current, new faceapi.TinyFaceDetectorOptions())
          .withFaceLandmarks()

        if (detections.length === 0) {
          onViolation('no_face')
        } else if (detections.length > 1) {
          onViolation('multiple_faces')
        } else {
          // Gaze check via nose tip vs face center
          const landmarks = detections[0].landmarks
          const nose = landmarks.getNose()[3]
          const leftEye = landmarks.getLeftEye()[0]
          const rightEye = landmarks.getRightEye()[3]
          const faceCenter = { x: (leftEye.x + rightEye.x) / 2, y: (leftEye.y + rightEye.y) / 2 }
          const gazeOffset = Math.abs(nose.x - faceCenter.x)
          if (gazeOffset > 30) onViolation('gaze_off')
        }
      }, 1000)

      // Tab visibility detection
      document.addEventListener('visibilitychange', () => {
        if (document.hidden) onViolation('tab_switch')
      })
    }
    init()
    return () => clearInterval(intervalRef.current)
  }, [onViolation])

  return { videoRef }
}

'use client'
import { useEffect, useState } from 'react'
import { createBrowserClient } from '@supabase/ssr'

export default function RecruiterSessionPage({ params }) {
  const [transcript, setTranscript] = useState([])
  const [evaluation, setEvaluation] = useState(null)
  const [warnings, setWarnings] = useState([])
  const supabase = createBrowserClient(...)

  useEffect(() => {
    // Subscribe to live messages
    supabase.channel(`messages-${params.id}`)
      .on('postgres_changes', { event: 'INSERT', schema: 'public', table: 'messages',
          filter: `session_id=eq.${params.id}` },
        (payload) => setTranscript(t => [...t, payload.new]))

      // Subscribe to evaluation ready event
      .on('broadcast', { event: 'evaluation-ready' },
        ({ payload }) => setEvaluation(payload))

      // Subscribe to proctoring events
      .on('postgres_changes', { event: 'INSERT', schema: 'public', table: 'proctor_events',
          filter: `session_id=eq.${params.id}` },
        (payload) => setWarnings(w => [...w, payload.new]))

      .subscribe()
  }, [params.id])

  return (
    <div className="grid grid-cols-3 gap-4 h-screen p-4">
      <LiveTranscriptPanel messages={transcript} />
      <ProctorEventLog events={warnings} />
      <EvaluationPanel evaluation={evaluation} />
    </div>
  )
}
